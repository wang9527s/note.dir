### **build_dropbear.sh**

```bash
#!/bin/bash

set -e

rm -rf dropbear-2024.84
tar -xjf dropbear-2024.84.tar.bz2

ToolChain="${HOME}/.toolchain/gcc-11.1.0-20210608-sigmastar-glibc-x86_64_arm-linux-gnueabihf/gcc-11.1.0-20210608-sigmastar-glibc-x86_64_arm-linux-gnueabihf"
export CC="${ToolChain}/bin/arm-linux-gnueabihf-gcc"
export CXX="${ToolChain}/bin/arm-linux-gnueabihf-g++"

rm -rf tmp-build && mkdir tmp-build
cd tmp-build
../dropbear-2024.84/configure --host=arm-linux-gnueabihf \
        -prefix=$(pwd)/../install \
        --disable-zlib \
        --enable-static

make -j$(nproc)
make install

echo "success"
```

### **start_dropbear.sh**

```bash
#!/bin/sh

ROOT_DIR="/mnt/sdcard/dropbear"
set -e

if [ ! -d "${ROOT_DIR}" ];then
        echo "ROOT_DIR: ${ROOT_DIR} not exist"
        exit -1
fi

cd "${ROOT_DIR}"

export LD_LIBRARY_PATH="${ROOT_DIR}/lib:$LD_LIBRARY_PATH"

if [ ! -d "${ROOT_DIR}/keys" ];then
    echo "create host_key"
    mkdir -p keys
    ./bin/dropbearkey -t rsa -f "${ROOT_DIR}/keys/dropbear_rsa_host_key"
    ./bin/dropbearkey -t ecdsa -f "${ROOT_DIR}/keys/dropbear_ecdsa_host_key"
    ./bin/dropbearkey -t ed25519 -f "${ROOT_DIR}/keys/dropbear_ed25519_host_key"
fi

kill -9 `pidof dropbear`

./sbin/dropbear -E -F -p 22 -r ./keys/dropbear_rsa_host_key > "./dropbear.log" 2>&1 &

echo "Dropbear SSH server started on port 22"
sleep 1
echo "pid is `pidof dropbear`"
```
